<script>
(function(){
  function getFilter(){ return new URLSearchParams(location.search).get('filter') || 'all'; }
  function apply(f){
    document.querySelectorAll('.pf-card').forEach(function(c){
      c.style.display = (f==='all' || c.dataset.aesthetic===f) ? '' : 'none';
    });
    document.querySelectorAll('.pf-filters .chip').forEach(function(b){
      b.classList.toggle('active', b.dataset.filter===f);
    });
  }
  document.querySelectorAll('.pf-filters .chip').forEach(function(b){
    b.addEventListener('click', function(){
      var f=b.dataset.filter, u=new URL(location.href);
      if(f==='all') u.searchParams.delete('filter'); else u.searchParams.set('filter', f);
      history.pushState({}, '', u); apply(f);
    });
  });
  apply(getFilter());
})();
</script>
