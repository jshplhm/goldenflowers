<script>
const cards = document.querySelectorAll('.wedding-card');
const sections = document.querySelectorAll('.wedding-section');
const emptyState = document.getElementById('empty-state');
const resultCount = document.getElementById('result-count');

let currentAesthetic = 'all';
let currentVenue = 'all';

document.querySelectorAll('.filter-chip').forEach(chip => {
  chip.addEventListener('click', function() {
    const filterType = this.dataset.filter;
    const val = this.dataset.value;
    // Update active state for this group
    document.querySelectorAll(`.filter-chip[data-filter="${filterType}"]`).forEach(c => c.classList.remove('active'));
    this.classList.add('active');
    if (filterType === 'aesthetic') currentAesthetic = val;
    if (filterType === 'venue') currentVenue = val;
    applyFilters();
  });
});

function applyFilters() {
  let visible = 0;
  cards.forEach(card => {
    const aMatch = currentAesthetic === 'all' || card.dataset.aesthetic === currentAesthetic;
    const vMatch = currentVenue === 'all' || card.dataset.venue === currentVenue;
    if (aMatch && vMatch) {
      card.classList.remove('filtered-out');
      visible++;
    } else {
      card.classList.add('filtered-out');
    }
  });

  // Show/hide sections
  sections.forEach(section => {
    const sectionAesthetic = section.dataset.section;
    const sectionCards = section.querySelectorAll('.wedding-card:not(.filtered-out)');
    if (sectionCards.length === 0 || (currentAesthetic !== 'all' && currentAesthetic !== sectionAesthetic)) {
      section.classList.add('hidden');
    } else {
      section.classList.remove('hidden');
    }
  });

  resultCount.textContent = visible + (visible === 1 ? ' wedding' : ' weddings');
  emptyState.style.display = visible === 0 ? 'block' : 'none';
}

// Read URL params on load
const params = new URLSearchParams(window.location.search);
const filterParam = params.get('filter');
const venueParam = params.get('venue');
if (filterParam) {
  const btn = document.querySelector(`.filter-chip[data-filter="aesthetic"][data-value="${filterParam}"]`);
  if (btn) btn.click();
}
if (venueParam) {
  const btn = document.querySelector(`.filter-chip[data-filter="venue"][data-value="${venueParam}"]`);
  if (btn) btn.click();
}
</script>
